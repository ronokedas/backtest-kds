//@version=5
strategy("Estrategia Didi Index Agulhada", overlay=true, default_qty_type=strategy.cash, default_qty_value=100, commission_type=strategy.commission.percent, commission_value=0.1)
sma3 = ta.sma(close, 3)
sma8 = ta.sma(close, 8)
sma20 = ta.sma(close, 20)
didiRapida = (sma3 - sma8) / sma8 * 100
didiLenta = (sma20 - sma8) / sma8 * 100
agulhadaAlta = ta.crossover(didiRapida, 0) and (didiLenta < 0) and (close > sma8)
agulhadaBaixa = ta.crossunder(didiRapida, 0) and (didiLenta > 0) and (close < sma8)
atr = ta.atr(14)

var float sl = na
var float tp = na
if agulhadaAlta and strategy.position_size == 0
    sl := close - atr * 1.5
    tp := close + atr * 2.5
    strategy.entry("Long Didi", strategy.long)
    strategy.exit("Exit Long", "Long Didi", stop=sl, limit=tp)
if agulhadaBaixa and strategy.position_size == 0
    sl := close + atr * 1.5
    tp := close - atr * 2.5
    strategy.entry("Short Didi", strategy.short)
    strategy.exit("Exit Short", "Short Didi", stop=sl, limit=tp)
plot(sma8, "SMA 8 Eixo", color=color.white)
